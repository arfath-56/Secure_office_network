! R1 config - my packet tracer lab
! (password replaced with <lab-password> for github)

hostname R1
no ip domain-lookup
ip domain-name lab.local

! one cable to SW1, so each vlan gets a subinterface (router on a stick)
interface gi0/0
 no shutdown
interface gi0/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
interface gi0/0.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip access-group GUEST_IN in
interface gi0/0.30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
interface gi0/0.99
 encapsulation dot1Q 99 native
 ip address 192.168.99.1 255.255.255.0

! guests could ping the server before this, so i blocked guest -> staff/servers/mgmt
! put it inbound on the guest subinterface so it gets dropped right away
ip access-list extended GUEST_IN
 deny ip 192.168.20.0 0.0.0.255 192.168.30.0 0.0.0.255
 deny ip 192.168.20.0 0.0.0.255 192.168.10.0 0.0.0.255
 deny ip 192.168.20.0 0.0.0.255 192.168.99.0 0.0.0.255
 permit ip any any

! locking down remote login
enable secret <lab-password>
username admin secret <lab-password>
service password-encryption
banner motd #Authorized access only. Activity is monitored.#

! had to type just "crypto key generate rsa" and then 1024, this IOS rejected "modulus"
crypto key generate rsa
ip ssh version 2

! only the staff network can ssh in, telnet is off
ip access-list standard VTY_ACCESS
 permit 192.168.10.0 0.0.0.255
line vty 0 4
 transport input ssh
 login local
 access-class VTY_ACCESS in
 exec-timeout 5 0
line console 0
 login local
 exec-timeout 5 0
